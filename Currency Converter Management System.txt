// Convert string to uppercase
void toUpperCase(char str[]) {
    for (int i = 0; str[i]; i++)
        str[i] = toupper(str[i]);
}

// Find currency index by code (returns -1 if not found)
int findCurrencyIndex(char code[]) {
    for (int i = 0; i < MAX_CURRENCIES; i++) {
        if (strcmp(code, currencies[i].code) == 0)
            return i;
    }
    return -1;
}

// Print spaces for formatting
void printSpaces(int count) {
    for (int i = 0; i < count; i++)
        cout << " ";
}

}

// ============ MENU OPTION 1: CONVERT CURRENCY ============

void convertCurrency() {
    char srcCode[STR_LEN];
    char tgtCode[STR_LEN];
    double amount;

    cout << "\n" << COLOR_CYAN << "======================================================\n";
    cout << "           CURRENCY CONVERSION\n";
    cout << "======================================================\n" << COLOR_RESET;

    // Display available currencies
    displayCurrencies();

    // Get source currency
    cout << COLOR_YELLOW << "Enter source currency code (e.g., PKR or AED): " << COLOR_RESET;
    cin >> srcCode;
    toUpperCase(srcCode);

    int srcIndex = findCurrencyIndex(srcCode);
    if (srcIndex == -1) {
        cout << COLOR_RED << "ERROR: Source currency '" << srcCode << "' not found!\n" << COLOR_RESET;
        return;
    }

    cout << COLOR_GREEN << "Source: " << srcCode << " (" << currencies[srcIndex].country << ")\n" << COLOR_RESET;

    // Get target currency
    cout << COLOR_YELLOW << "Enter target currency code (e.g., USD or SAR): " << COLOR_RESET;
    cin >> tgtCode;
    toUpperCase(tgtCode);

    int tgtIndex = findCurrencyIndex(tgtCode);
    if (tgtIndex == -1) {
        cout << COLOR_RED << "ERROR: Target currency '" << tgtCode << "' not found!\n" << COLOR_RESET;
        return;
    }

    cout << COLOR_GREEN << "Target: " << tgtCode << " (" << currencies[tgtIndex].country << ")\n" << COLOR_RESET;

    // Get amount
    cout << COLOR_YELLOW << "Enter amount to convert: " << COLOR_RESET;
    cin >> amount;

    // Input validation
    if (cin.fail()) {
        cin.clear();
        cin.ignore(10000, '\n');
        cout << COLOR_RED << "ERROR: Invalid input! Please enter a valid number.\n" << COLOR_RESET;
        return;
    }

    // Check for negative amount
    if (amount < 0) {
        cout << COLOR_RED << "ERROR: Amount cannot be negative!\n" << COLOR_RESET;
        return;
    }

    // Calculate conversion
    double conversionRate = currencies[tgtIndex].rate / currencies[srcIndex].rate;
    double result = amount * conversionRate;

    // Display result
    cout << "\n" << COLOR_CYAN << "======================================================\n";
    cout << "                CONVERSION RESULT\n";
    cout << "======================================================\n" << COLOR_RESET;
    cout << COLOR_GREEN << amount << " " << srcCode << COLOR_RESET << " = " << COLOR_MAGENTA << result << " " << tgtCode << COLOR_RESET << "\n\n";
    cout << COLOR_YELLOW << "Exchange Rate: 1 " << srcCode << " = " << conversionRate << " " << tgtCode << COLOR_RESET << "\n";
    cout << COLOR_CYAN << "======================================================\n" << COLOR_RESET;

    // Save to history
    if (historyCount < MAX_HISTORY) {
        history[historyCount].amount = amount;
        strcpy(history[historyCount].fromCurrency, srcCode);
        strcpy(history[historyCount].toCurrency, tgtCode);
        history[historyCount].result = result;
        history[historyCount].rateUsed = conversionRate;
        historyCount++;
        cout << COLOR_GREEN << "Conversion saved to history.\n" << COLOR_RESET;
    } else {
        cout << COLOR_RED << "WARNING: History buffer is full!\n" << COLOR_RESET;
    }
}

// ============ MENU OPTION 4: UPDATE EXCHANGE RATE (ADMIN) ============

void updateExchangeRate() {
    char password[STR_LEN];
    char currencyCode[STR_LEN];
    double newRate;

    cout << "\n" << COLOR_CYAN << "======================================================\n";
    cout << "      UPDATE EXCHANGE RATE (ADMIN ONLY)\n";
    cout << "======================================================\n" << COLOR_RESET;
    cout << COLOR_YELLOW << "This function requires admin password.\n" << COLOR_RESET;

    // Password verification
    cout << COLOR_YELLOW << "Enter admin password: " << COLOR_RESET;
    cin >> password;

    if (strcmp(password, ADMIN_PASSWORD) != 0) {
        cout << COLOR_RED << "ERROR: Incorrect password! Access denied.\n" << COLOR_RESET;
        return;
    }

    cout << COLOR_GREEN << "Password correct. Access granted.\n" << COLOR_RESET;

    // Display available currencies
    displayCurrencies();

    // Get currency to update
    cout << COLOR_YELLOW << "Enter currency code to update: " << COLOR_RESET;
    cin >> currencyCode;
    toUpperCase(currencyCode);

    int index = findCurrencyIndex(currencyCode);
    if (index == -1) {
        cout << COLOR_RED << "ERROR: Currency '" << currencyCode << "' not found!\n" << COLOR_RESET;
        return;
    }

    // Prevent USD modification
    if (strcmp(currencyCode, "USD") == 0) {
        cout << COLOR_RED << "ERROR: Cannot modify the base currency (USD)!\n" << COLOR_RESET;
        return;
    }

    // Display current rate
    cout << "\n" << COLOR_BLUE << "Current rate for " << currencyCode 
         << " (" << currencies[index].country << "): "
         << currencies[index].rate << COLOR_RESET << "\n";

    // Get new rate
    cout << COLOR_YELLOW << "Enter new exchange rate: " << COLOR_RESET;
    cin >> newRate;

    // Input validation - FIXED BUG
    if (cin.fail()) {
        cin.clear();
        cin.ignore(10000, '\n');
        cout << COLOR_RED << "ERROR: Invalid input! Please enter a valid number.\n" << COLOR_RESET;
        return;
    }

    if (newRate < 0) {
        cout << COLOR_RED << "ERROR: Exchange rate cannot be negative!\n" << COLOR_RESET;
        return;
    }

    if (newRate == 0) {
        cout << COLOR_RED << "ERROR: Exchange rate cannot be zero!\n" << COLOR_RESET;
        return;
    }

    // Update the rate
    double oldRate = currencies[index].rate;
    currencies[index].rate = newRate;

    cout << "\n" << COLOR_CYAN << "======================================================\n";
    cout << "      RATE UPDATED SUCCESSFULLY!\n";
    cout << "======================================================\n" << COLOR_RESET;
    cout << COLOR_GREEN << "Currency: " << currencyCode << " (" << currencies[index].country << ")\n";
    cout << "Old Rate: " << oldRate << "\n";
    cout << "New Rate: " << newRate << COLOR_RESET << "\n";
    cout << COLOR_CYAN << "======================================================\n" << COLOR_RESET;
}

// ============ MENU OPTION 5: VIEW CONVERSION HISTORY ============

void viewConversionHistory() {
    cout << "\n" << COLOR_CYAN << "======================================================\n";
    cout << "           CONVERSION HISTORY\n";
    cout << "======================================================\n" << COLOR_RESET;

    if (historyCount == 0) {
        cout << COLOR_YELLOW << "No conversion history available.\n" << COLOR_RESET;
        return;
    }

    cout << COLOR_GREEN << "Total conversions: " << historyCount << COLOR_RESET << "\n\n";

    cout << COLOR_BLUE << "From    To      Amount          Result          Rate\n";
    cout << "----    --      ------          ------          ----\n" << COLOR_RESET;

    for (int i = 0; i < historyCount; i++) {
        cout << COLOR_GREEN << history[i].fromCurrency << COLOR_RESET;
        printSpaces(7 - strlen(history[i].fromCurrency));
        
        cout << COLOR_GREEN << history[i].toCurrency << COLOR_RESET;
        printSpaces(7 - strlen(history[i].toCurrency));
        
        cout << history[i].amount;
        printSpaces(15 - (int)strlen("") - (history[i].amount < 10 ? 1 : (history[i].amount < 100 ? 2 : 3)));
        
        cout << COLOR_MAGENTA << history[i].result << COLOR_RESET;
        printSpaces(15 - (int)strlen(""));
        
        cout << COLOR_YELLOW << history[i].rateUsed << COLOR_RESET << "\n";
    }
    cout << COLOR_CYAN << "======================================================\n" << COLOR_RESET;
}
